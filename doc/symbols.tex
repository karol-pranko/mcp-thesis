\section*{B. Wykaz symboli}

\begin{longtable}[l]{>{\bfseries\raggedright\arraybackslash}p{2.8cm} >{\raggedright\arraybackslash}p{12.0cm}}
$B_{\text{tok}}$ & Konfigurowalny budżet tokenów dla pojedynczej kwerendy \\[1.5pt]
$B_{\text{net}}$ & Przepustowość łącza sieciowego w~modelu transferu Out-of-Band \\[1.5pt]
$v_x$ & Przepustowość operacji $x$ (kodowanie, dekodowanie, SHA-256, kopiowanie) w~modelu transferu Out-of-Band \\[1.5pt]
$d$ & Wymiarowość przestrzeni osadzeń wektorowych ($d = 384$ dla modelu \texttt{all-MiniLM-L6-v2}) \\[1.5pt]
$\delta_x$ & Stały narzut trybu transferu $x$, niezależny od rozmiaru ładunku \\[1.5pt]
$h_{\text{KV}}$ & Liczba głów kluczy i~wartości w~mechanizmie uwagi (GQA) \\[1.5pt]
$K$ & Rozmiar strony pobieranej w~procedurze paginacji kursorowej (liczba wierszy) \\[1.5pt]
$k_{\text{RRF}}$ & Stała stabilizująca fuzję rang Reciprocal Rank Fusion ($k_{\text{RRF}} = 60$) \\[1.5pt]
$M$ & Liczba współbieżnych klientów (zadań asyncio w~jednym procesie) w~Scenariuszu C1 \\[1.5pt]
$M_{\text{exp}}$ & Maksymalna liczba tabel mostkujących dodawanych przez ekspansję grafową ($M_{\text{exp}} = 3$) \\[1.5pt]
$\mu_n, \sigma_n$ & Bieżąca estymacja średniej oraz odchylenia standardowego średnich paczkowych liczby tokenów na wiersz w~algorytmie Welforda \\[1.5pt]
$N$ & Liczebność zbioru ewaluacyjnego Text-to-SQL ($N = 500$) lub liczba rekordów w~tabeli analitycznej ($N = 1\ 000\ 000$) \\[1.5pt]
$S$ & Rozmiar ładunku binarnego w~bajtach \\[1.5pt]
$S^*(\text{RTT})$ & Punkt równowagi (próg opłacalności) transferu Out-of-Band w~funkcji opóźnienia sieciowego \\[1.5pt]
$S^*_{\text{SHM}}$ & Próg opłacalności transferu przez pamięć współdzieloną zmierzony na jednym węźle ($\wynSstarLoopShm$~KiB) \\[1.5pt]
$R_{\text{res}}$ & Odsetek rekordów z~pozostałymi tokenami sterującymi po enkapsulacji XML \\[1.5pt]
$W, U, p, r$ & Statystyki testów Wilcoxona ($W$) i~Manna--Whitneya ($U$), wartość $p$ oraz korelacja rangowo-biserialna $r$ \\
\end{longtable}
