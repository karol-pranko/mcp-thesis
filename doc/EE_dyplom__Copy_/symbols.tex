\section*{B. Wykaz Symboli i Oznaczeń Matematycznych}

\begin{longtable}[l]{>{\bfseries\raggedright\arraybackslash}p{2.8cm} >{\raggedright\arraybackslash}p{12.0cm}}
$B_{\text{tok}}$ & Konfigurowalny budżet tokenów dla pojedynczej kwerendy \\[1.5pt]
$B_{\text{net}}, B_{\text{shm}}$ & Przepustowość odpowiednio łącza sieciowego oraz magistrali pamięci współdzielonej \\[1.5pt]
$d$ & Wymiarowość przestrzeni osadzeń wektorowych ($d = 384$ dla modelu \texttt{all-MiniLM-L6-v2}) \\[1.5pt]
$\delta_{\text{OOB}}$ & Stały narzut przygotowania transferu poza pasmem (alokacja deskryptora, skrót kryptograficzny) \\[1.5pt]
$K$ & Rozmiar strony pobieranej w~procedurze paginacji kursorowej (liczba wierszy) \\[1.5pt]
$k_{\text{RRF}}$ & Stała stabilizująca fuzję rang Reciprocal Rank Fusion ($k_{\text{RRF}} = 60$) \\[1.5pt]
$M$ & Liczba współbieżnych procesów agentowych / wątków klienckich \\[1.5pt]
$M_{\text{exp}}$ & Maksymalna dopuszczalna liczba wstrzykiwanych tabel mostkujących ($M_{\text{exp}} = 3$) \\[1.5pt]
$\mu_n, \sigma_n$ & Bieżąca estymacja średniej oraz odchylenia standardowego liczby tokenów na rekord w~algorytmie Welforda \\[1.5pt]
$N$ & Liczebność zbioru ewaluacyjnego Text-to-SQL ($N = 500$) lub liczba rekordów w~tabeli analitycznej ($N = 1\ 000\ 000$) \\[1.5pt]
$S$ & Rozmiar ładunku binarnego w~bajtach \\[1.5pt]
$S^*(\text{RTT})$ & Punkt równowagi (próg opłacalności) transferu Out-of-Band w~funkcji opóźnienia sieciowego \\[1.5pt]
$W, p, r$ & Statystyka testowa, poziom istotności oraz standaryzowana wielkość efektu w~teście sumy rangowych znaków Wilcoxona \\
\end{longtable}
