[complex, |GT| = 4] tabele: finance_currency_hedges_quarterly, finance_cash_flow_projections_domestic, finance_ledger_entries_daily, finance_accounts_payable_international
Pytanie: Sporządź zaawansowane zapytanie analityczne CTE bazujące na tabeli finance_currency_hedges_quarterly zwracające jawne kolumny: b.base_id (finance_currency_hedges_quarterly.id), b.status, b.metric_val (wartość agregacji), finance_cash_flow_projections_domestic_id (public.finance_cash_flow_projections_domestic.id), finance_ledger_entries_daily_id (public.finance_ledger_entries_daily.id), finance_accounts_payable_international_id (public.finance_accounts_payable_international.id), integrujące relacje z tabelami (finance_cash_flow_projections_domestic, finance_ledger_entries_daily, finance_accounts_payable_international), filtrując b.metric_val IS NOT NULL z sortowaniem ORDER BY b.base_id ASC (LIMIT 50).
SQL wzorcowy: WITH base_summary AS (
    SELECT public.finance_currency_hedges_quarterly.id AS base_id, public.finance_currency_hedges_quarterly.status, SUM(public.finance_currency_hedges_quarterly.amount) AS metric_val
    FROM public.finance_currency_hedges_quarterly
    WHERE public.finance_currency_hedges_quarterly.id > 0
    GROUP BY public.finance_currency_hedges_quarterly.id, public.finance_currency_hedges_quarterly.status
)
SELECT b.base_id, b.status, b.metric_val, public.finance_cash_flow_projections_domestic.id AS finance_cash_flow_projections_domestic_id, public.finance_ledger_entries_daily.id AS finance_ledger_entries_daily_id, public.finance_accounts_payable_international.id AS finance_accounts_payable_international_id
FROM base_summary b
JOIN public.finance_currency_hedges_quarterly ON b.base_id = public.finance_currency_hedges_quarterly.id
JOIN public.finance_cash_flow_projections_domestic ON public.finance_currency_hedges_quarterly.id = public.finance_cash_flow_projections_domestic.fk_currency_id JOIN public.finance_ledger_entries_daily ON public.finance_cash_flow_projections_domestic.fk_ledger_id = public.finance_ledger_entries_daily.id JOIN public.finance_accounts_payable_international ON public.finance_ledger_entries_daily.fk_accounts_id = public.finance_accounts_payable_international.id
WHERE b.metric_val IS NOT NULL
ORDER BY b.base_id ASC
LIMIT 50;
